## 1. Project Setup

### 1.1 Initialize Vite + SolidJS + TypeScript

```bash
# From project root
cd assistant/backend
npm create vite@latest frontend -- --template solid-ts
# Accept all defaults; this creates: frontend/{package.json, index.html, src/main.tsx}
```

### 1.2 Install Dependencies

```bash
cd frontend
npm install \
  solid-router \
  @solidjs/meta \
  zod \
  vitest \
  @testing-library/solidjs \
  @types/node \
  @types/testing-library__solidjs \
  --save-dev
```

### 1.3 Configure TypeScript Strict Mode

Edit `frontend/tsconfig.json`:
```json
{
  "compilerOptions": {
    "target": "ES2022",
    "module": "ESNext",
    "lib": ["ES2022", "DOM", "DOM.Iterable"],
    "jsx": "preserve",
    "strict": true,
    "noUncheckedIndexedAccess": true,
    "noImplicitAny": true,
    "strictNullChecks": true,
    "esModuleInterop": true,
    "moduleResolution": "bundler",
    "allowImportingExtensions": true,
    "forceConsistentCasingInFileNames": true,
    "isolatedModules": true,
    "declaration": true,
    "outDir": "dist"
  },
  "include": ["src"],
  "exclude": ["node_modules", "dist"]
}
```

### 1.4 Configure Vite Config

Edit `frontend/vite.config.ts`:
```typescript
import { defineConfig } from 'vite'
import solid from 'vite-plugin-solid'

export default defineConfig({
  plugins: [solid()],
  base: '/static/',
  build: {
    target: 'es2022',
    sourcemap: true,
    minify: 'esbuild',
    rollupOptions: {
      output: {
        manualChunks: {
          solid: ['solid-js', 'solid-router'],
          ui: ['@solidjs/meta']
        }
      }
    }
  },
  server: {
    port: 5173,
    host: true
  }
})
```

### 1.5 Output to Correct Directory

Update `frontend/package.json` `build` script or add to `vite.config.ts`:
```typescript
// vite.config.ts build output
export default defineConfig({
  // ...
  build: {
    // Output to assistant/backend/static so Caddy serves it
    outDir: '../assistant/backend/static',
    emptyOutDir: true
  }
})
```

Or alternatively, keep `outDir: dist` and add a post-build copy step in Dockerfile.

## 3. Core Requirements

```typescript

createRoot(document.getElementById('root')!).render(<App />)
```

### 3.5 `src/state/chat.ts` — Chat State Management

```typescript
import { createSignal, createStore } from 'solid-js'

export interface ChatMessage {
  role: 'user' | 'assistant'
  content: string
  id: string
  meta?: MessageMeta
}

export interface MessageMeta {
  task_type?: string
  memory_context?: string
  citations?: string[]
}

export const [messages, setMessages] = createSignal<ChatMessage[]>([])
export const [queue, setQueue] = createSignal<QueuedMessage[]>([])
export const [isTurnActive, setTurnActive] = createSignal(false)
export const [sessionId, setSessionId] = createSignal<string | null>(null)

// Load from localStorage on init
if (typeof localStorage !== 'undefined') {
  const saved = localStorage.getItem('session_id')
  if (saved) setSessionId(saved)
}
```

### 3.6 `src/services/storage.ts` — localStorage Helpers

```typescript
export const saveState = <T>(key: string, state: T) => {
  if (typeof localStorage !== 'undefined') {
    localStorage.setItem(key, JSON.stringify(state))
  }
}

export const loadState = <T>(key: string, defaultValue: T): T => {
  if (typeof localStorage !== 'undefined') {
    const saved = localStorage.getItem(key)
    return saved ? JSON.parse(saved) : defaultValue
  }
  return defaultValue
}
```

### 3.7 `src/components/ui/ThemeToggle.tsx` — Light/Dark Mode

```typescript
import { createSignal, } from 'solid-js'

const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches

const [isDark, setIsDark] = createSignal(prefersDark)

export const ThemeToggle = () => {
  const toggle = () => setIsDark(prev => !prev)

  // Initialize on mount
  createEffect(() => {
    if (isDark()) {
      document.documentElement.classList.add('dark')
    } else {
      document.documentElement.classList.remove('dark')
    }
  })

  return (
    <button
      className="theme-toggle"
      onClick={toggle}
      aria-label={isDark() ? 'Switch to light mode' : 'Switch to dark mode'}
    >
      {isDark() ? '☀️' : '🌙'}
    </button>
  )
}
```

## 9. Agent Notes

- **Template used**: `vite --template solid-ts` generates a working SolidJS + TypeScript project
- **React mental model**: SolidJS ≈ React JSX syntax; agents familiar with React can read TSX immediately
- **Signals vs VDOM**: SolidJS signals are granular; reading/writing a signal triggers precise reactive updates — no re-renders of unchanged components
- **TypeScript strict mode**: Catches the exact errors agents currently make (null access, wrong props, missing fields)
- **Zod schemas**: Validate at runtime AND serve as TypeScript types — single source of truth for API contract
- **Testing**: Vitest + `@testing-library/solidjs` runs tests in < 50ms per file; fast feedback loop
- **Docker build impact**: +30 seconds to build time; negligible operational cost
